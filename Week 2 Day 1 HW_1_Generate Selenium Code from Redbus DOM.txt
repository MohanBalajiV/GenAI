import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;
import org.testng.annotations.AfterClass;
import org.testng.annotations.BeforeClass;
import org.testng.annotations.Test;

import java.time.Duration;

/**
 * Test class for ticket booking application search functionality.
 */
public class TicketBookingSearchTest {

    private WebDriver driver;
    private WebDriverWait wait;

    /**
     * Sets up the WebDriver and WebDriverWait before tests.
     */
    @BeforeClass
    public void setUp() {
        driver = new ChromeDriver();
        wait = new WebDriverWait(driver, Duration.ofSeconds(10));
        driver.manage().window().maximize();
    }

    /**
     * Performs bus search for ticket booking.
     */
    @Test
    public void performBusSearch() {
        navigateToSearchPage();
        enterSourceCity("New York");
        enterDestinationCity("Boston");
        selectJourneyDate();
        toggleWomenBooking();
        clickSearchBusesButton();
    }

    /**
     * Navigates to the search page of the application.
     */
    private void navigateToSearchPage() {
        driver.get("https://www.ticketbooking.com"); // Replace with actual URL
    }

    /**
     * Enters the source city in the search form.
     *
     * @param sourceCity The source city to enter.
     */
    private void enterSourceCity(String sourceCity) {
        WebElement sourceField = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//div[@class='srcDestWrapper___11b20c' and .//div[text()='From']]")));
        sourceField.click();
        WebElement inputField = wait.until(ExpectedConditions.visibilityOfElementLocated(
                By.xpath("//div[@class='srcDestWrapper___11b20c' and .//div[text()='From']]//input")));
        inputField.sendKeys(sourceCity);
    }

    /**
     * Enters the destination city in the search form.
     *
     * @param destinationCity The destination city to enter.
     */
    private void enterDestinationCity(String destinationCity) {
        WebElement destinationField = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//div[@class='srcDestWrapper___11b20c' and .//div[text()='To']]")));
        destinationField.click();
        WebElement inputField = wait.until(ExpectedConditions.visibilityOfElementLocated(
                By.xpath("//div[@class='srcDestWrapper___11b20c' and .//div[text()='To']]//input")));
        inputField.sendKeys(destinationCity);
    }

    /**
     * Selects the journey date from the date picker.
     */
    private void selectJourneyDate() {
        WebElement dateField = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//div[@class='dateInputWrapper___50eff3' and @aria-label='Select Date of Journey. Current date: 23 Oct, 2025']")));
        dateField.click();
        WebElement todayButton = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//button[@aria-label='Search for Today' and text()='Today']")));
        todayButton.click();
    }

    /**
     * Toggles the women booking option.
     */
    private void toggleWomenBooking() {
        WebElement womenBookingSwitch = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//input[@id='switch' and @type='checkbox']")));
        womenBookingSwitch.click();
    }

    /**
     * Clicks the search buses button to initiate the search.
     */
    private void clickSearchBusesButton() {
        WebElement searchButton = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//button[@class='primaryButton___93b44e searchButton___1e2c20' and .//i[contains(@class, 'icon-search')]]")));
        searchButton.click();
    }

    /**
     * Cleans up resources after tests.
     */
    @AfterClass
    public void tearDown() {
        if (driver != null) {
            driver.quit();
        }
    }
}
