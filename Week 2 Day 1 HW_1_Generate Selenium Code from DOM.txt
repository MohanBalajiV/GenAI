import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;
import org.testng.annotations.AfterClass;
import org.testng.annotations.BeforeClass;
import org.testng.annotations.Test;

import java.time.Duration;

/**
 * Test class for ticket booking application login functionality.
 */
public class TicketBookingLoginTest {

    private WebDriver driver;
    private WebDriverWait wait;

    /**
     * Sets up the WebDriver and WebDriverWait before tests.
     */
    @BeforeClass
    public void setUp() {
        driver = new ChromeDriver();
        wait = new WebDriverWait(driver, Duration.ofSeconds(10));
        driver.manage().window().maximize();
    }

    /**
     * Performs login to the ticket booking application.
     */
    @Test
    public void performLogin() {
        navigateToLoginPage();
        enterUserId("testUser");
        enterPassword("testPassword");
        clickLoginButton();
    }

    /**
     * Navigates to the login page of the application.
     */
    private void navigateToLoginPage() {
        driver.get("https://www4.phillipmobile.com.sg"); // Replace with actual URL
    }

    /**
     * Enters the user ID in the login form.
     *
     * @param userId The user ID to enter.
     */
    private void enterUserId(String userId) {
        WebElement userIdField = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("userid")));
        userIdField.clear();
        userIdField.sendKeys(userId);
    }

    /**
     * Enters the password in the login form.
     *
     * @param password The password to enter.
     */
    private void enterPassword(String password) {
        WebElement passwordField = wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("password")));
        passwordField.clear();
        passwordField.sendKeys(password);
    }

    /**
     * Clicks the login button to submit the login form.
     */
    private void clickLoginButton() {
        WebElement loginButton = wait.until(ExpectedConditions.elementToBeClickable(By.id("btn-login-submit")));
        loginButton.click();
    }

    /**
     * Clicks the Forgot/Change Password link.
     */
    @Test(dependsOnMethods = "performLogin")
    public void clickForgotPasswordLink() {
        WebElement forgotPasswordLink = wait.until(ExpectedConditions.elementToBeClickable(
                By.xpath("//a[@class='forgot-pass login-text' and contains(text(), 'Forget/Change Password')]")));
        forgotPasswordLink.click();
    }

    /**
     * Cleans up resources after tests.
     */
    @AfterClass
    public void tearDown() {
        if (driver != null) {
            driver.quit();
        }
    }
}
