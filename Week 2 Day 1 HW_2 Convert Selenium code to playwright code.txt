    }

import { Page, expect } from '@playwright/test';

/**
 * Represents the login page of the Trading application.
 * Provides methods to interact with login form elements and perform login actions.
 */
export class TradingLoginPage {
  private page: Page;

  /**
   * Constructs a new TradingLoginPage instance.
   *
   * @param page The Playwright Page instance used to interact with the browser
   */
  constructor(page: Page) {
    this.page = page;
  }

  /**
   * Enters the provided user ID into the login ID field.
   *
   * @param userId The user ID to be entered
   */
  async enterUserId(userId: string): Promise<void> {
    await this.page.locator('#userid').waitFor({ state: 'visible', timeout: 60000 });
    await this.page.locator('#userid').clear();
    await this.page.locator('#userid').fill(userId);
  }

  /**
   * Enters the provided password into the password field.
   *
   * @param password The password to be entered
   */
  async enterPassword(password: string): Promise<void> {
    await this.page.locator('#password').waitFor({ state: 'visible', timeout: 60000 });
    await this.page.locator('#password').clear();
    await this.page.locator('#password').fill(password);
  }

  /**
   * Clicks the Login button to submit the login form.
   */
  async clickLoginButton(): Promise<void> {
    await this.page.locator('#btn-login-submit').waitFor({ state: 'visible', timeout: 60000 });
    await this.page.locator('#btn-login-submit').click();
  }

  /**
   * Performs a complete login action using the provided user ID and password.
   *
   * @param userId The user ID for authentication
   * @param password The password for authentication
   */
  async login(userId: string, password: string): Promise<void> {
    await this.enterUserId(userId);
    await this.enterPassword(password);
    await this.clickLoginButton();
  }

  /**
   * Verifies if the login error message is displayed on the page.
   *
   * @returns True if the error message is present, false otherwise
   */
  async isLoginErrorMessageDisplayed(): Promise<boolean> {
    try {
      await this.page
        .locator("//div[contains(@class, 'login-error-summary')]//li[text()='Incorrect login. Please try again.']")
        .waitFor({ state: 'visible', timeout: 60000 });
      return await this.page
        .locator("//div[contains(@class, 'login-error-summary')]//li[text()='Incorrect login. Please try again.']")
        .isVisible();
    } catch (e) {
      return false;
    }
  }

  /**
   * Clicks the 'Forget/Change Password' link.
   */
  async clickForgotPasswordLink(): Promise<void> {
    await this.page
      .locator("//a[@class='forgot-pass login-text' and text()='Forget/Change Password']")
      .waitFor({ state: 'visible', timeout: 60000 });
    await this.page
      .locator("//a[@class='forgot-pass login-text' and text()='Forget/Change Password']")
      .click();
  }
}
